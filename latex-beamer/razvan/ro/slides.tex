\section{Introducere}

\begin{frame}{Motivație}
  \textbf{Problema:}
  \begin{itemize}
    \item Recomandări personalizate $\rightarrow$ date stocate central
    \item Utilizatorul alege: \textit{confidențialitate} sau \textit{personalizare}
  \end{itemize}

  \vskip0.8em
  \textbf{Consecințe:}
  \begin{itemize}
    \item Locații GPS, preferințe $\rightarrow$ profil comportamental pe server
    \item Fără personalizare reală $\rightarrow$ popularitate amplifică aglomerarea
  \end{itemize}

  \vskip0.8em
  \textbf{Propunere:} confidențialitate și personalizare \textit{nu se exclud:} \\
  \hspace{1em}antrenare locală pe dispozitiv
\end{frame}

\begin{frame}{State of the art}
  \textbf{Aplicații de explorare urbană:}
  \begin{itemize}
    \item Google Maps, TripAdvisor: personalizare centralizată
    \item izi.TRAVEL, GetYourGuide: rute fixe, fără personalizare
    \item Niciuna: rute personalizate + privacy by design + gamificare
  \end{itemize}

  \vskip0.6em
  \textbf{Federated Learning (FL) în recomandări:}
  \begin{itemize}
    \item FedAvg: standard de facto; cere clienți simultan disponibili
    \item FedAsync: asincron cu discount de vechime
    \item DP (NbAFL): zgomot Gaussian
  \end{itemize}

  \vskip0.6em
  \textbf{Golul:} FL asincron cu DP pe turism cultural, neabordat
\end{frame}

\section{Soluție propusă}

\begin{frame}{Soluție propusă}
  Patru componente integrate:
  \begin{itemize}
    \item Obiective culturale pe hartă
    \item Quest-uri prin geofencing ($<$ 100 m)
    \item Rute generate dinamic, personalizate prin FL
    \item Comunitate: recenzii, povești, propuneri de obiective, moderare
  \end{itemize}

  \vskip0.8em
  \textbf{Privacy by design:} interesele sunt învățate local din interacțiuni, fără ca datele să părăsească dispozitivul
\end{frame}

\section{Personalizare și rute}

\begin{frame}{Modelul de personalizare}
  \textbf{MLP} $22 \rightarrow 32 \rightarrow 16 \rightarrow 1$; \quad 1\,281 parametri; inferență $\approx$ 39\,µs pe laptop

  \vskip0.4em
  \begin{center}
  \scriptsize
  \begin{tabular}{ll}
    \textbf{Grup (dim.)} & \textbf{Conținut} \\
    \hline
    Interese {[}0–5{]}   & artă, arhit., istorie, gastron., natură, muzică \\
    Tip {[}6–13{]}       & muzeu, monum., parc, galerie, restaur., piață, clădire, altele \\
    Temporal {[}14–16{]} & isOpen, isWeekend, ora/24 \\
    Spațial {[}17–21{]}  & rang dist., potrivire interese, în rută, poziție, lungime \\
  \end{tabular}
  \end{center}

  \vskip0.4em
  \normalsize
  \textbf{Ieșire:} scor $\in [0, 1]$ $\rightarrow$ scăzut ($< 0{,}4$) · mediu · ridicat ($\geq 0{,}7$) \\
  {\footnotesize Rulează pe \textbf{server} (evaluare candidați) și pe \textbf{dispozitiv} (scor popup)}
\end{frame}

\begin{frame}{Generarea rutelor personalizate}
  \textbf{Algoritmul (pe server):}
  \begin{enumerate}
    \item Candidați: obiective în \textbf{raza 10 km}
    \item Scoring: \textbf{80\% MLP + 20\% proximitate} $\rightarrow$ top 10
    \item Traseu \textbf{greedy}: $\min(\text{dist} - \lambda \cdot \text{match})$, $\lambda$ configurat de utilizator
    \item Durata per oprire: scalată $\pm 25\%$ de scorul MLP
    \item Traseu real: \textbf{OSRM} (walking / driving)
  \end{enumerate}
\end{frame}

\begin{frame}{Pre-antrenarea modelului global}
  \textbf{Problema cold-start:} la prima utilizare, niciun dispozitiv nu a contribuit încă la modelul global

  \vskip0.6em
  \begin{center}
  \scriptsize
  \begin{tabular}{ll}
    \hline
    Pre-antrenare (741k înreg., Foursquare, Yelp)          & {[}0,595; 0,850{]}; blocat la medie \\
    Fine-tuning (+16k sintetice + 3$\times$ scoruri $\leq 0{,}4$) & {[}0,050; 0,841{]}; \textbf{+211\%} \\
    \hline
  \end{tabular}
  \end{center}

  \vskip0.5em
  \normalsize
  Fără fine-tuning: predicție 0,70–0,75 indiferent de input \\
  {\small $\rightarrow$ etichete concentrate spre valori mari (Yelp: 60\% recenzii 4–5 stele; Foursquare: fix 0,70) + \texttt{isOpen=0} absent}
\end{frame}

\section{Federated Learning}

\begin{frame}{Antrenare locală}
  \textbf{Etichete buffer:}
  \begin{center}
  \scriptsize
  \begin{tabular}{ll}
    \textbf{Eveniment} & \textbf{Etichetă} \\
    \hline
    Sheet obiectiv deschis & 0,3 \\
    Adăugat la rută / Quest încercat & 0,5 \\
    Navigație pornită      & 0,6 \\
    Quest sugerat          & 0,80 \\
    Poveste trimisă        & 0,85 \\
    Quest completat        & 0,9 \\
    Rating explicit (stele) & stele / 5 \\
  \end{tabular}
  \end{center}

  \vskip0.6em
  \textbf{Procedura (FedAvg client):}
  \begin{itemize}
    \item Declanșare automată după 15 interacțiuni în buffer
    \item 5 epoci SGD (lr = 0,01), gradient clipping L2 = 1
    \item Buffer persistat în SharedPreferences
    \item Trimite doar delta ponderilor, nu interacțiunile brute
  \end{itemize}
\end{frame}

\begin{frame}{Agregare asincronă, concurență și confidențialitate}
  \textbf{FedAsync:} delta ponderată cu $\frac{1}{1+s}$; clienți „stale" contribuie mai puțin

  \vskip0.6em
  \textbf{Redis lock:} \texttt{SET NX PX} $\rightarrow$ un singur client în ciclul read–aggregate–write; TTL = 5 s

  \vskip0.6em
  \textbf{Differential Privacy (NbAFL):}
  \begin{itemize}
    \item \textbf{Client:} clamp $[-1, +1]$ + $\mathcal{N}(0,\,0{,}01)$ per coordonată
    \item \textbf{Server:} clipping L2 $\leq 1$ pe delta primită
  \end{itemize}
\end{frame}

\section{Concluzii}

\begin{frame}{Limitări}
  \begin{itemize}
    \item \textbf{Validare pe clienți sintetici:} comportamentul în producție rămâne neverificat pe date reale
    \item \textbf{Proximitate exclusiv în foreground:} GPS se oprește în background; soluție: geofencing nativ OS
    \item \textbf{Scalabilitate Redis lock:} 8/20 cereri FL simultane respinse; soluție: coadă de lucru (Redis Queue)
    \item \textbf{FedPer exclus:} straturi personalizate per utilizator necesită stocare locală permanentă
  \end{itemize}
\end{frame}

\begin{frame}{Concluzii}
  \textbf{Ce am construit:} aplicație Flutter end-to-end cu flux FL complet (pre-antrenare $\rightarrow$ FedAsync + DP)

  \vskip0.6em
  \textbf{Ce am obținut:}
  \begin{itemize}
    \item Rute personalizate diferite per utilizator; quests prin geofencing
    \item Comunitate activă: recenzii, povești, propuneri de obiective
    \item Interese învățate local; datele nu părăsesc dispozitivul
  \end{itemize}

  \vskip0.6em
  \textbf{Ce urmează:} validare pe utilizatori reali; FedPer; FedBuff (agregare cu buffer); fotografii obiective
\end{frame}

\section{Backup}

\begin{frame}{Algoritmul de generare a rutei}
  \textbf{1. Filtrare candidați} (raza 10 km)\\
  {\small scor inițial $= \text{FL} \times 0{,}8 + (1 - d_{\text{norm}}) \times 0{,}2$ $\rightarrow$ top $2n$ candidați}

  \vskip0.5em
  \textbf{2. Greedy nearest-neighbour}\\
  {\small La fiecare pas: $\displaystyle \operatorname{argmin}_{l} \bigl[\operatorname{dist}(\text{crt}, l) - \lambda \cdot \operatorname{match}(l)\bigr]$}\\
  {\small $\operatorname{match}(l)$ = fracție categorii obiectiv $\cap$ interese utilizator; $\lambda$ configurat de utilizator}\\
  {\small Se oprește când bugetul de timp e depășit sau s-a atins $n_{\max}$ opriri}

  \vskip0.5em
  \textbf{3. Scalare timp vizită cu MLP}\\
  {\small $t = t_{\text{baza}} \times (0{,}75 + 0{,}5 \times \text{FL\_score})$; $t_{\text{baza}}$ pe tip: muzeu 70 min, monument 20 min \ldots}

  \vskip0.5em
  \textbf{4. Traseu real via OSRM} (walking / driving)
\end{frame}

\begin{frame}{Cold-start: ce vede un utilizator nou?}
  \textbf{1. Onboarding} $\rightarrow$ utilizatorul selectează interesele (artă, arhitectură, \ldots)\\
  {\small Intră direct în vectorul de caracteristici [0–5]; nicio interacțiune necesară}

  \vskip0.6em
  \textbf{2. Model global descărcat} (GET \texttt{/model/global})\\
  {\small Model pre-antrenat pe Foursquare + Yelp; recomandările sunt utile de la prima sesiune}

  \vskip0.6em
  \textbf{3. Buffer de interacțiuni se umple}\\
  {\small 1 intrare / obiectiv / rundă (deduplicat); după \textbf{15 interacțiuni} $\rightarrow$ rundă FL declanșată automat}

  \vskip0.6em
  \textbf{4. Prima contribuție FL}\\
  {\small Antrenare locală pe dispozitiv $\rightarrow$ delta trimis la server $\rightarrow$ model actualizat; personalizarea se construiește progresiv}
\end{frame}

\begin{frame}{Taxonomia FL și poziționarea CultureQuest}
  \textbf{Tipul de partiție a datelor:}
  \begin{itemize}
    \item \textbf{HFL} (Orizontal): toți clienții antrenează același model pe același spațiu de 22 caracteristici; diferă doar interacțiunile locale
    \item VFL și FTL: excluse; nu există suprapunere de caracteristici diferite între clienți
  \end{itemize}

  \vskip0.5em
  \textbf{Tipul de desfășurare:}
  \begin{itemize}
    \item \textbf{Cross-Device}: mii de telefoane, conectate intermitent, cu volum mic de date locale
    \item Cross-Silo: exclus; nu există organizații cu date partajate
  \end{itemize}

  \vskip0.5em
  CultureQuest: \textbf{HFL Cross-Device asincron} cu DP (NbAFL)
\end{frame}

\begin{frame}{De ce FedAsync și nu FedProx / SCAFFOLD / FedDyn?}
  FedProx, SCAFFOLD, FedDyn presupun o \textbf{cohortă sincronă} de clienți per rundă.\\
  {\small CultureQuest procesează un client pe rând $\rightarrow$ nu există cohortă $\rightarrow$ mecanismele de corecție nu au referință.}

  \vskip0.6em
  \begin{center}
  \scriptsize
  \begin{tabular}{lccc}
    \textbf{Algoritm} & \textbf{Sincron} & \textbf{Mecanism} & \textbf{Compatibil} \\
    \hline
    FedAvg   & Da  & Medie ponderată              & Da (local) \\
    FedAsync & Nu  & Discount staleness $1/(1+s)$ & \textbf{Da} \\
    FedProx  & Da  & Termen proximal              & Nu \\
    SCAFFOLD & Da  & Variabilă de control         & Nu \\
    FedDyn   & Da  & Regularizator dinamic        & Nu \\
  \end{tabular}
  \end{center}
\end{frame}

\begin{frame}{Differential Privacy în detaliu}
  \textbf{Mecanismul DP} (NbAFL, pe client):
  \begin{enumerate}
    \item Clampare per coordonată: $\delta_i \in [-\tau, \tau]$, $\tau = 1$ $\rightarrow$ mărginește sensibilitatea
    \item Zgomot Gaussian: $\tilde{w} = w_{\text{global}} + \mathrm{clip}(\delta,\tau) + \mathcal{N}(0,\,(\sigma\tau)^2)$, $\sigma = 0{,}1$
  \end{enumerate}

  \vskip0.5em
  \textbf{Mecanisme corelate} (scopuri diferite):
  \begin{itemize}
    \item \textbf{Client, în SGD:} limitare L2 gradient $\|g\|_2 \leq 1$ $\rightarrow$ stabilitate antrenare, nu DP
    \item \textbf{Server, la agregare:} limitare L2 deltă $\|\delta\|_2 \leq \tau$ $\rightarrow$ robustețe față de actualizări adversariale, nu DP
  \end{itemize}

  \vskip0.5em
  {\small Zgomotul adăugat pe client $\rightarrow$ serverul primește ponderi din care delta reală nu mai poate fi reconstituită}
\end{frame}

\begin{frame}{Scalabilitate: testul de concurență FL}
  \begin{center}
  \scriptsize
  \begin{tabular}{ccccc}
    \textbf{Cereri simultane} & \textbf{Succes} & \textbf{Erori} & \textbf{Lat. medie (ms)} & \textbf{Lat. p95 (ms)} \\
    \hline
    1  & 1  & 0 & 6,5     & 6,5    \\
    5  & 5  & 0 & 622,6   & 923,5  \\
    10 & 10 & 0 & 1.384,1 & 2.442,2 \\
    20 & 12 & 8 & 2.064,9 & 3.076,1 \\
  \end{tabular}
  \end{center}

  \vskip0.5em
  \textbf{Concluzie:} lock Redis $\rightarrow$ 0 erori până la 10 cereri simultane\\
  La 20: 8/20 cereri respinse (TTL lock expirat); fără corupere date\\
  {\small Soluție: coadă (Redis Queue) $\rightarrow$ elimină lock-ul din calea HTTP}
\end{frame}

\begin{frame}{Consum de baterie și proximitate în foreground}
  \textbf{Curent:} geofencing manual (\texttt{geolocator}); GPS activ doar în foreground\\
  {\small $\rightarrow$ quest-urile nu se activează când aplicația e în background}

  \vskip0.8em
  \textbf{Alternativă:} geofencing nativ OS
  \begin{itemize}
    \item iOS: \textit{Region Monitoring}; Android: \textit{Geofence API}
    \item OS gestionează GPS; notifică doar la traversarea razei $\rightarrow$ consum redus
  \end{itemize}
\end{frame}
